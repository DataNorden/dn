export function createPathFilter(
  callback: (ins: Omit<PredicateBuilder<PathPredicate>, 'compile'>) => void | unknown,
  defaultGlobOptions?: MinimatchOptions,
): PathPredicate {
  const ins = new PredicateBuilder<PathPredicate>(
    new PathPredicateStrategy(),
    defaultGlobOptions,
  );

  callback(ins);
  return ins.compile();
}

export function createPathStatFilter<T extends FsPathType = FsPathType>(

  arg1?:
    | MinimatchOptions
    | ((ins: Omit<PredicateBuilder<PathStatPredicate>, 'compile'>) => void | unknown),
  arg2?: MinimatchOptions,
) {
  const types = (Array.isArray(arg0) ? arg0 : []) as T[];
  const callback = (Array.isArray(arg0) ? arg1! : arg0) as (
    ins: Omit<PredicateBuilder<PathStatPredicate>, 'compile'>,
  ) => void | unknown;
  const defaultGlobOptions = (Array.isArray(arg0) ? arg2 : arg1) as MinimatchOptions;

  const ins = new PredicateBuilder<PathStatPredicate>(
    new PathStatPredicateStrategy<T>(types),
    defaultGlobOptions,
  );
  callback(ins);
  return ins.compile();
}
